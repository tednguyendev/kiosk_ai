🔧 Local Proxy Server (Node.js)

If CORS blocks direct calls, run this proxy:

// proxy.js — run with: node proxy.js
const express = require('express');
const cors = require('cors');
const fetch = require('node-fetch');
const app = express();
app.use(cors());
app.use(express.json());

app.post('/api/bithuman/:agentCode/speak', async (req, res) => {
  const r = await fetch(`https://api.bithuman.ai/v1/agent/${req.params.agentCode}/speak`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'api-secret': req.headers['x-api-secret']
    },
    body: JSON.stringify(req.body)
  });
  const data = await r.json();
  res.status(r.status).json(data);
});

app.listen(3456, () => console.log('Proxy on http://localhost:3456'));
